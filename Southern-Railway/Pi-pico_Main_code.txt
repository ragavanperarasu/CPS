#define PBAT 27 // Primary Battery pin

#define MAINVOLT 28 // 110 volt detection (500 > variable => present 110 volt)

//Global Variable

int BBatValue = 0;
// Add these with your other global variables
unsigned long displaySwitchTime = 0;
const unsigned long DISPLAY_SWITCH_INTERVAL = 10000; // 2 seconds

bool MaintenanceValue = false;
bool MainVolt = false;

int RXPin = 13;  // for gps
int TXPin = 12;

int signalQuality;

double latGPS = 0.0;
double lonGPS = 0.0;

SoftwareSerial gpsSerial(RXPin, TXPin);

  startUp();

void loop() {
  unsigned long gpsStart = millis();
  while (millis() - gpsStart < 500) {
    while (gpsSerial.available() > 0) {
      if (gps.encode(gpsSerial.read())) {
        displayInfo();
      }
    }
  }
  
  handleSignal();
  voltageDetection();
  handleMaintenanceSwitch();
  handleMainVolt();
  handleRelay();
  preparePayload();
  //sendHTTPDataToServer();

  if (millis() - lastSendTime >= 60000) { // 60000 ms = 1 minute
  if(signalQuality > 35 || signalQuality < 15){
      lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("  GSM Modem  ");
  lcd.setCursor(0, 1);  // Move to second line
  lcd.print("Restart...");
  modem.sendAT("AT+CFUN=1, 1");
  delay(5000);
  lcd.clear();
  }else{
    sendHTTPDataToServer();
  }
    
    lastSendTime = millis(); // reset timer
  }
}

void startUp(){
  lcd.begin(16, 2);
  lcd.print(" FDS MONITORING");
  lcd.setCursor(0, 1);  // Move to second line
  lcd.print("     SYSTEM");
  delay(2000);
  lcd.setCursor(0, 1);  // Move to second line
  lcd.print("Initializing...");
  
  SerialMon.begin(115200);
  
  delay(3000);

  gpsSerial.begin(9600);

  delay(1000);

  pinMode(MAINTENANCE, INPUT_PULLUP);
  pinMode(RELEYSWITCH, OUTPUT);
  
  
  Serial.println("*STARTING FDS MONITORING SYSTEM BY GCT");
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("  GSM Modem  ");
  lcd.setCursor(0, 1);  // Move to second line
  lcd.print("Initializing...");
  
  initModem();

  lcd.setCursor(0, 0);
  lcd.print("  GSM Network  ");
  lcd.setCursor(0, 1);  // Move to second line
  lcd.print("Connecting...");
  
  if (!connectToNetwork()) return;
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print(" GPRS Network ");
  lcd.setCursor(0, 1);  // Move to second line
  lcd.print("Connecting...");
  if (!connectToGPRS()) return;
  lcd.clear();
}

void readGPSData() {
 while (gpsSerial.available() > 0) {
    if (gps.encode(gpsSerial.read())) {
      displayInfo();
    }
  }
  if (millis() > 5000 && gps.charsProcessed() < 10) {
    Serial.println("No GPS data received");
  }
}

void displayInfo() {
  if (gps.location.isValid()) {
    latGPS = gps.location.lat();
    lonGPS = gps.location.lng();
  } else {
    Serial.println("[ WARN ] Location Not available");
  }
}


void initModem() {
  //Serial.println("Initializing Modem...");
  SerialAT.begin(115200);
  delay(3000);

  if (!modem.init()) {
    Serial.println("[ PROBLEM ] Failed to initialize modem");
      lcd.setCursor(0, 1);  // Move to second line
      lcd.print("Init Failed...");
  }

  if (GSM_PIN && modem.getSimStatus() != 3) {
    modem.simUnlock(GSM_PIN);
  }

  Serial.println("[ STATUS ] Modem initialized Success");
  lcd.setCursor(0, 1);  // Move to second line
  lcd.print("Successfully...");
  delay(2000);
  lcd.clear();
}

bool connectToNetwork() {
  if (!modem.waitForNetwork()) {
    Serial.println("[ PROBLEM ] GSM Network connection failed");
    lcd.setCursor(0, 1);  // Move to second line
    lcd.print("     Failed    ");
    return false;
  }
  Serial.println("[ STATUS ] GSM Network connected");
  lcd.setCursor(0, 1);  // Move to second line
    lcd.print(" Successfully  ");
  return true;
}

bool connectToGPRS() {
  //Serial.println("Connecting to GPRS...");
  if (!modem.gprsConnect(apn, gprsUser, gprsPass)) {
    Serial.println("[ PROBLEM ] GPRS connection failed");
    lcd.setCursor(0, 1);  // Move to second line
  lcd.print("     Failed    ");
    return false;
  }
  Serial.println("[ STATUS ] GPRS connected");
  lcd.setCursor(0, 1);
  lcd.print(" Successfully  ");
  return true;
}

void preparePayload() {
  payloadObj["coachid"] = "B7-203813";
  payloadObj["pribat"] = PBatValue;
  payloadObj["backbat"] = BBatValue;
  payloadObj["pripow"] = MainVolt;
  payloadObj["maintainance"] = MaintenanceValue;
  payloadObj["lat"] = latGPS;
  payloadObj["lng"] = lonGPS;
  payloadObj["sig"] = signalQuality;
  
  serializeJson(payloadObj, payload);
  Serial.print("📡[ STATUS ] ");
  Serial.println(payload);



  lcd.setCursor(0, 0);
  lcd.print("PB BB M SS HV OP");
  
  lcd.setCursor(0, 1);  // Move to second line
  lcd.print(PBatValue);

  lcd.setCursor(3, 1);  // Move to second line
  lcd.print(BBatValue);

  lcd.setCursor(8, 1);  // Move to second line
  lcd.print(signalQuality);

  if(MaintenanceValue == true){
    lcd.setCursor(6, 1);  // Move to second line
  lcd.print("T");
  }
  else{
    lcd.setCursor(6, 1);  // Move to second line
  lcd.print("F");
  }

  if(MainVolt == true){
    lcd.setCursor(11, 1);  // Move to second line
  lcd.print("T");
  }
  else{
    lcd.setCursor(11, 1);  // Move to second line
  lcd.print("F");
  }

  if (millis() - displaySwitchTime >= DISPLAY_SWITCH_INTERVAL) {
     lcd.clear();  
    displaySwitchTime = millis();
    lcd.setCursor(0, 0);
    lcd.print("LAT:");
      lcd.print(latGPS, 4); // Show 4 decimal places
      
      // Show longitude on second line
      lcd.setCursor(0, 1);
      lcd.print("LNG:");
      lcd.print(lonGPS, 4);

      delay(4000);


    lcd.clear(); // Clear screen when switching
  }




  
}

void sendHTTPDataToServer() {
  Serial.println("[ STATUS ]📡 Making HTTP POST request...");
    lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print(" Send MQTT Req ");
  
  lcd.setCursor(0, 1);
  lcd.print("    Sending ....");

  
  

  mqttClient.setServer(mqttBroker, mqttPort);
  
  // Connect to MQTT
  Serial.print("Connecting to MQTT...");
  String clientId = "pico-" + String(random(1000, 9999));
  
  if (mqttClient.connect(clientId.c_str())) {
    Serial.println("Connected");
    
    // Send single test message
    //String message = "{\"test\":\"Hello from Srinath Pico\",\"value\":123}";
 if (mqttClient.publish(mqttTopic, payload.c_str())) {
      Serial.println("Message sent: " + payload);
    } else {
      Serial.println("Publish failed");
    }
    
  } else {
    Serial.println("MQTT connect failed");
  }
  
  // Disconnect
  delay(2000);
 
   lcd.clear();
}



bool sendATCommand(String cmd, long timeout) {
  modem.sendAT(cmd);
  if (modem.waitResponse(timeout) != 1) {
    //Serial.println("❌ AT command failed: " + cmd);
    return false;
  }
  return true;
}

void handleMaintenanceSwitch(){
  MaintenanceValue = digitalRead(MAINTENANCE);
}

void handleMainVolt(){
  int MainVoltAna = analogRead(MAINVOLT);
  if(MainVoltAna < 500)
  {
    MainVolt = true;
  }
  else
  {
    MainVolt = false;
  }
}

void handleRelay(){

    Serial.print("PBatValue: ");
  Serial.print(PBatValue);
  Serial.print(", Reley State: ");

  if(PBatValue < 22){
    digitalWrite(RELEYSWITCH, HIGH);
    lcd.setCursor(14, 1);  // Move to second line
    lcd.print("BB");
       Serial.println("ON (BB)");
  
  }
  else{
    digitalWrite(RELEYSWITCH, LOW);
    lcd.setCursor(14, 1);  // Move to second line
    lcd.print("PB");
       Serial.println("ON (pb)");
  }
}
